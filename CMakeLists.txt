cmake_minimum_required(VERSION 3.20)
# Base release version (MAJOR.MINOR.PATCH); CI may release a higher patch,
# see tools/release-version.sh.
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" MTDD_BASE_VERSION LIMIT_COUNT 1)
project(mtdd VERSION ${MTDD_BASE_VERSION} LANGUAGES CXX)
# What `mtdd --version` and mtdd_build_info report. CI sets it to the release
# version (X.Y.Z) or to dev-<short sha>; local builds say "<base>-local".
set(MTDD_BUILD_VERSION "${PROJECT_VERSION}-local" CACHE STRING "Version string embedded in mtdd")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(MTDD_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
option(MTDD_COVERAGE "Enable GCC/Clang coverage instrumentation" OFF)
include(CTest)
find_package(OpenSSL 3 REQUIRED)
find_package(Threads REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl>=7.85)
pkg_check_modules(LIBEVENT REQUIRED IMPORTED_TARGET libevent libevent_pthreads)

add_library(mtdd_options INTERFACE)
target_compile_options(mtdd_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(MTDD_SANITIZE)
  target_compile_options(mtdd_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(mtdd_options INTERFACE -fsanitize=address,undefined)
endif()
if(MTDD_COVERAGE)
  target_compile_options(mtdd_options INTERFACE --coverage)
  target_link_options(mtdd_options INTERFACE --coverage)
endif()

add_library(mtdd_config src/config.cpp src/metadata.cpp)
target_include_directories(mtdd_config PUBLIC include)
target_link_libraries(mtdd_config PUBLIC mtdd_options nlohmann_json::nlohmann_json PkgConfig::CURL Threads::Threads)

# Component ownership is isolated; CONFIGURE_DEPENDS also discovers workers' sources.
foreach(component protocol control io)
  file(GLOB component_sources CONFIGURE_DEPENDS "src/${component}/*.cpp")
  if(component_sources)
    add_library(mtdd_${component} ${component_sources})
    target_include_directories(mtdd_${component} PUBLIC include)
    target_link_libraries(mtdd_${component} PUBLIC mtdd_options Threads::Threads)
    if(NOT component STREQUAL "io")
      target_link_libraries(mtdd_${component} PUBLIC OpenSSL::Crypto)
    endif()
    if(component STREQUAL "control")
      target_link_libraries(mtdd_${component} PUBLIC nlohmann_json::nlohmann_json PkgConfig::LIBEVENT)
    endif()
  endif()
endforeach()

if(TARGET mtdd_protocol AND TARGET mtdd_io)
  add_executable(mtdd_me_probe tools/me_probe.cpp)
  target_link_libraries(mtdd_me_probe PRIVATE mtdd_config mtdd_protocol mtdd_io)
endif()

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/runtime.cpp")
  file(GLOB runtime_sources CONFIGURE_DEPENDS "src/runtime/*.cpp")
  add_library(mtdd_runtime src/runtime.cpp ${runtime_sources})
  target_include_directories(mtdd_runtime PUBLIC include)
  target_link_libraries(mtdd_runtime PUBLIC mtdd_config mtdd_protocol mtdd_control mtdd_io)
endif()

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp" AND TARGET mtdd_runtime)
  add_executable(mtdd src/main.cpp)
  target_link_libraries(mtdd PRIVATE mtdd_runtime)
  target_compile_definitions(mtdd PRIVATE MTDD_VERSION="${MTDD_BUILD_VERSION}")
  install(TARGETS mtdd RUNTIME DESTINATION bin)
endif()

if(BUILD_TESTING)
  find_package(GTest REQUIRED)
  include(GoogleTest)
  foreach(component config protocol control io runtime)
    set(test_source "tests/unit/${component}_test.cpp")
    if(TARGET mtdd_${component} AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${test_source}")
      add_executable(${component}_tests ${test_source})
      if(component STREQUAL "runtime")
        file(GLOB runtime_test_sources CONFIGURE_DEPENDS "tests/unit/runtime_cases_*.cpp")
        target_sources(runtime_tests PRIVATE ${runtime_test_sources})
      elseif(component STREQUAL "control")
        file(GLOB control_test_sources CONFIGURE_DEPENDS "tests/unit/control_cases_*.cpp")
        target_sources(control_tests PRIVATE ${control_test_sources})
      elseif(component STREQUAL "protocol")
        file(GLOB protocol_test_sources CONFIGURE_DEPENDS "tests/unit/protocol_cases_*.cpp")
        target_sources(protocol_tests PRIVATE ${protocol_test_sources})
      endif()
      target_link_libraries(${component}_tests PRIVATE mtdd_${component} GTest::gtest_main)
      gtest_discover_tests(${component}_tests DISCOVERY_TIMEOUT 30 PROPERTIES TIMEOUT 30)
    endif()
  endforeach()
  if(TARGET mtdd_control AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/unit/ip_history_test.cpp")
    add_executable(ip_history_tests tests/unit/ip_history_test.cpp)
    target_link_libraries(ip_history_tests PRIVATE mtdd_control GTest::gtest_main)
    gtest_discover_tests(ip_history_tests DISCOVERY_TIMEOUT 30 PROPERTIES TIMEOUT 30)
  endif()
  add_executable(runtime_support_tests tests/unit/runtime_support_test.cpp)
  add_executable(fake_tls_send_tests tests/unit/fake_tls_send_test.cpp)
  target_link_libraries(fake_tls_send_tests PRIVATE mtdd_protocol GTest::gtest_main)
  gtest_discover_tests(fake_tls_send_tests PROPERTIES TIMEOUT 30)
  add_executable(retry_tests tests/unit/retry_test.cpp)
  target_link_libraries(retry_tests PRIVATE mtdd_options GTest::gtest_main)
  gtest_discover_tests(retry_tests PROPERTIES TIMEOUT 30)
  # Header-only ME orphan-close policy: deterministic, no sockets or sessions.
  add_executable(me_control_tests tests/unit/me_control_test.cpp)
  target_link_libraries(me_control_tests PRIVATE mtdd_options GTest::gtest_main)
  gtest_discover_tests(me_control_tests PROPERTIES TIMEOUT 30)
  add_executable(me_audit_tests tests/unit/me_audit_test.cpp)
  target_link_libraries(me_audit_tests PRIVATE mtdd_runtime GTest::gtest_main)
  gtest_discover_tests(me_audit_tests DISCOVERY_TIMEOUT 30 PROPERTIES TIMEOUT 30)
  target_include_directories(runtime_support_tests PRIVATE include)
  target_link_libraries(runtime_support_tests PRIVATE mtdd_options Threads::Threads GTest::gtest_main)
  gtest_discover_tests(runtime_support_tests PROPERTIES TIMEOUT 30)
endif()
